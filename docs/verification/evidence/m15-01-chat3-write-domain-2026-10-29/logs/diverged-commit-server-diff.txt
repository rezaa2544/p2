+  const uids = [...new Set((ops || []).filter(op => op && op.uid).map(op => String(op.uid)))].sort();
+  return transaction(async client => {
+    for (const uid of uids) {
+      await client.query('SELECT pg_advisory_xact_lock(hashtextextended($1, 0))', [uid]);
